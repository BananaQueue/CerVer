# LeafCode PoC Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Prove whether CerVer can encode a per-page payload into a bespoke leaf-shaped 2D symbology and decode it reliably from an image, via automated feasibility gates — without touching the shipping leaf-QR.

**Architecture:** Pure-JS, isolated `src/leafcode/` modules: a GF(256) Reed–Solomon codec, a deterministic 256-node leaf lattice, an encoder (payload→bits) and its inverse, a pure-JS rasterizer (test double of the visual), and an image decoder (threshold → blob detection → 3-anchor affine rectify → node sampling → RS decode). Gate A (clean round-trip) and Gate B (synthetic degradation) run as automated `node:test` suites using the rasterizer, so no browser or camera is needed to make the go/no-go call.

**Tech Stack:** Node.js 24 built-ins only (`node:test`, `node:assert`). No new dependencies. Canvas `ImageData` is represented as a plain `{ width, height, data: Uint8ClampedArray }` object.

## Global Constraints

- Node.js **24.x**; `node:test` + `node:assert/strict` only. **No new npm dependencies.**
- Everything lives under `src/leafcode/` and `test/leafcode/`. **No changes** to the sealer, verify flow, API, or the shipping leaf-QR.
- ESM modules (`"type": "module"` already set).
- Symbology fixed values: **256 nodes**, **3 scalene triangular anchors**, **1 bit/node** (solid=1 / hollow=0), **32-byte RS codeword = 13 data + 19 parity** (corrects up to 9 byte errors).
- Canonical lattice coordinate space: **0..1000 × 0..1000** (`SPACE = 1000`).
- Determinism: lattice PRNG seed is the fixed constant **`0x1EAF`**; the lattice must be identical across runs.
- Gate A is a hard kill-switch: **100%** clean round-trip or stop.

```js
// GF(2^8) with primitive polynomial 0x11d, generator 2.
export const EXP = new Uint8Array(512);
export const LOG = new Uint8Array(256);
(() => {
  let x = 1;
  for (let i = 0; i < 255; i++) {
    EXP[i] = x;
    LOG[x] = i;
    x <<= 1;
    if (x & 0x100) x ^= 0x11d;
  }
  for (let i = 255; i < 512; i++) EXP[i] = EXP[i - 255];
})();

export function mul(a, b) {
  if (a === 0 || b === 0) return 0;
  return EXP[LOG[a] + LOG[b]];
}
export function inv(a) {
  if (a === 0) throw new Error('inv(0)');
  return EXP[255 - LOG[a]];
}
```

```js
import { EXP, LOG, mul, inv } from './gf256.js';

const polyMul = (a, b) => {
  const r = new Uint8Array(a.length + b.length - 1);
  for (let i = 0; i < a.length; i++)
    for (let j = 0; j < b.length; j++) r[i + j] ^= mul(a[i], b[j]);
  return r;
};
const polyEval = (p, x) => {
  let y = 0;
  for (let i = 0; i < p.length; i++) y = mul(y, x) ^ p[i];
  return y;
};
function generator(nsym) {
  let g = Uint8Array.from([1]);
  for (let i = 0; i < nsym; i++) g = polyMul(g, Uint8Array.from([1, EXP[i]]));
  return g;
}

export function rsEncode(data, nsym) {
  const gen = generator(nsym);
  const out = new Uint8Array(data.length + nsym);
  out.set(data, 0);
  for (let i = 0; i < data.length; i++) {
    const coef = out[i];
    if (coef !== 0)
      for (let j = 1; j < gen.length; j++) out[i + j] ^= mul(gen[j], coef);
  }
  out.set(data, 0); // restore data region (systematic)
  return out;
}

function syndromes(code, nsym) {
  const s = new Uint8Array(nsym);
  for (let i = 0; i < nsym; i++) s[i] = polyEval(code, EXP[i]);
  return s;
}

export function rsDecode(code, nsym) {
  code = Uint8Array.from(code);
  const synd = syndromes(code, nsym);
  if (synd.every((v) => v === 0)) return code.slice(0, code.length - nsym);

  // Berlekamp–Massey → error locator polynomial
  let errLoc = Uint8Array.from([1]);
  let oldLoc = Uint8Array.from([1]);
  for (let i = 0; i < nsym; i++) {
    oldLoc = Uint8Array.from([...oldLoc, 0]);
    let delta = synd[i];
    for (let j = 1; j < errLoc.length; j++) delta ^= mul(errLoc[errLoc.length - 1 - j], synd[i - j]);
    if (delta !== 0) {
      if (oldLoc.length > errLoc.length) {
        let newLoc = oldLoc.map((c) => mul(c, delta));
        oldLoc = errLoc.map((c) => mul(c, inv(delta)));
        errLoc = newLoc;
      }
      const scaled = oldLoc.map((c) => mul(c, delta));
      const res = new Uint8Array(Math.max(errLoc.length, scaled.length));
      for (let k = 0; k < errLoc.length; k++) res[res.length - errLoc.length + k] = errLoc[k];
      for (let k = 0; k < scaled.length; k++) res[res.length - scaled.length + k] ^= scaled[k];
      errLoc = res;
    }
  }
  const errs = errLoc.length - 1;
  if (errs * 2 > nsym) return null;

  // Chien search → error positions
  const positions = [];
  for (let i = 0; i < code.length; i++) {
    if (polyEval(errLoc, EXP[255 - i]) === 0) positions.push(code.length - 1 - i);
  }
  if (positions.length !== errs) return null;

  // Forney → error magnitudes
  const revSynd = synd.slice().reverse();
  let errEval = revSynd;
  {
    const xnsym = new Uint8Array(nsym + 1);
    xnsym[0] = 1;
    // omega = (synd(rev) * errLoc) mod x^nsym
    const prod = polyMul(revSynd, errLoc);
    errEval = prod.slice(prod.length - nsym);
  }
  for (const pos of positions) {
    const xi = EXP[255 - (code.length - 1 - pos)];
    const xiInv = inv(xi);
    // formal derivative of errLoc, evaluated at xiInv
    let denom = 0;
    for (let k = 1; k < errLoc.length; k += 2) {
      // derivative in GF(2): keep odd-index terms
    }
    // Evaluate omega(xiInv)
    const num = polyEval(errEval, xiInv);
    // Evaluate errLoc' (odd terms) at xiInv
    let der = 0;
    const el = errLoc;
    for (let k = 1; k < el.length; k++) {
      if ((el.length - 1 - k) % 2 === 0) der ^= mul(el[k], EXP[(255 - (code.length - 1 - pos)) * ((el.length - 1 - k)) % 255]);
    }
    // Robust derivative: rebuild using standard method
    der = 0;
    for (let k = el.length - 2; k >= 0; k -= 2) der ^= el[k];
    let derVal = 0;
    // Evaluate derivative polynomial (even-powered coeffs) at xiInv
    {
      const deriv = [];
      for (let k = 0; k < el.length - 1; k++) {
        if ((el.length - 2 - k) % 2 === 0) deriv.push(el[k]);
        else deriv.push(0);
      }
      derVal = polyEval(Uint8Array.from(deriv), xiInv);
    }
    if (derVal === 0) return null;
    const mag = mul(num, inv(derVal));
    code[pos] ^= mul(mag, xi === 0 ? 1 : 1); // magnitude applied at position
    code[pos] ^= 0; // (magnitude already xored below)
    code[pos] ^= mag; // final correction
  }
  // Re-verify
  if (!syndromes(code, nsym).every((v) => v === 0)) return null;
  return code.slice(0, code.length - nsym);
}
```

```js
export const SPACE = 1000;
const SEED = 0x1eaf;

function mulberry32(a) {
  return function () {
    a |= 0;
    a = (a + 0x6d2b79f5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

// Leaf geometry in a centered space, then shifted into [0,SPACE].
const Lh = 820;   // leaf length
const Wmax = 250; // leaf max half-width
const CX = SPACE / 2;
const CY = SPACE / 2;

export function hw(u) {
  return (Wmax * Math.pow(Math.max(0, u), 0.45) * Math.pow(Math.max(0, 1 - u), 0.85)) / 0.45;
}
// u: 0 base -> 1 tip ; returns absolute canvas coords (tip toward top)
export function P(u, x) {
  return { x: CX + x, y: CY + (0.5 - u) * Lh };
}

export function lattice() {
  const rnd = mulberry32(SEED);
  const minD = 44;
  const nodes = [];
  let tries = 0;
  while (nodes.length < 256 && tries < 200000) {
    tries++;
    const u = 0.05 + rnd() * 0.9;
    const maxx = hw(u) * 0.9;
    const x = (rnd() * 2 - 1) * maxx;
    const p = P(u, x);
    let ok = true;
    for (const n of nodes) {
      const dx = n.x - p.x, dy = n.y - p.y;
      if (dx * dx + dy * dy < minD * minD) { ok = false; break; }
    }
    if (ok) nodes.push({ x: p.x, y: p.y, u });
  }
  if (nodes.length < 256) throw new Error(`lattice only placed ${nodes.length} nodes`);
  nodes.sort((a, b) => (a.u - b.u) || (a.x - b.x));
  const clean = nodes.map((n) => ({ x: n.x, y: n.y }));
  const anchors = [P(0.9, 0), P(0.3, -hw(0.3) * 0.6), P(0.09, 0)];
  return { nodes: clean, anchors, hw, P };
}
```

> **Tuning note:** if Step 4 throws "only placed N nodes", lower `minD` (e.g. 42)
> until ≥256 fit, then keep it fixed. The value must stay constant thereafter.

```js
import { rsEncode, rsDecode } from './rs.js';

const B32 = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ234567';
const HEADER = [0x1e, 0x01]; // magic, version
const NSYM = 19;

function packPayload(payload) {
  const m = payload.match(/^CVR\|R1-((?:19|20)\d\d)-(\d{6})\|(\d{1,3})\|([A-Z2-7]{4})-([A-Z2-7]{4})$/);
  if (!m) throw new Error('bad payload: ' + payload);
  const year = Number(m[1]);
  const serial = Number(m[2]);
  const page = Number(m[3]);
  const sealChars = (m[4] + m[5]).toUpperCase();
  let sealBits = 0n;
  for (const c of sealChars) sealBits = (sealBits << 5n) | BigInt(B32.indexOf(c)); // 40 bits
  const data = new Uint8Array(13);
  data[0] = HEADER[0];
  data[1] = HEADER[1];
  data[2] = (year >> 8) & 0xff;
  data[3] = year & 0xff;
  data[4] = (serial >> 16) & 0xff;
  data[5] = (serial >> 8) & 0xff;
  data[6] = serial & 0xff;
  data[7] = page & 0xff;
  for (let i = 0; i < 5; i++) data[8 + i] = Number((sealBits >> BigInt((4 - i) * 8)) & 0xffn);
  return data;
}

function unpackPayload(data) {
  if (data[0] !== HEADER[0] || data[1] !== HEADER[1]) return null;
  const year = (data[2] << 8) | data[3];
  const serial = (data[4] << 16) | (data[5] << 8) | data[6];
  const page = data[7];
  let sealBits = 0n;
  for (let i = 0; i < 5; i++) sealBits = (sealBits << 8n) | BigInt(data[8 + i]);
  let chars = '';
  for (let i = 7; i >= 0; i--) chars = B32[Number((sealBits >> BigInt(i * 5)) & 0x1fn)] + chars;
  const seal = chars.slice(0, 4) + '-' + chars.slice(4);
  const serialStr = String(serial).padStart(6, '0');
  return `CVR|R1-${year}-${serialStr}|${page}|${seal}`;
}

export function encode(payload) {
  const code = rsEncode(packPayload(payload), NSYM); // 32 bytes
  const bits = new Uint8Array(256);
  for (let i = 0; i < 32; i++)
    for (let b = 0; b < 8; b++) bits[i * 8 + b] = (code[i] >> (7 - b)) & 1;
  return bits;
}

export function bitsToPayload(bits) {
  const code = new Uint8Array(32);
  for (let i = 0; i < 32; i++) {
    let v = 0;
    for (let b = 0; b < 8; b++) v = (v << 1) | (bits[i * 8 + b] & 1);
    code[i] = v;
  }
  const data = rsDecode(code, NSYM);
  if (!data) return null;
  return unpackPayload(data);
}
```

```js
import { lattice, SPACE } from './lattice.js';

export function rasterize(bits, { px = 600, nodeR = 9, anchorR = 13 } = {}) {
  const { nodes, anchors } = lattice();
  const s = px / SPACE;
  const data = new Uint8ClampedArray(px * px * 4).fill(255);
  const set = (x, y, v) => {
    x = Math.round(x); y = Math.round(y);
    if (x < 0 || y < 0 || x >= px || y >= px) return;
    const o = (y * px + x) * 4;
    data[o] = data[o + 1] = data[o + 2] = v; data[o + 3] = 255;
  };
  const disc = (cx, cy, r, fill) => {
    for (let y = -r; y <= r; y++)
      for (let x = -r; x <= r; x++) {
        const d = Math.hypot(x, y);
        if (fill ? d <= r : d <= r && d >= r - 2.2) set(cx + x, cy + y, 0);
      }
  };
  const tri = (cx, cy, r) => {
    const pts = [[cx, cy - r], [cx - r * 0.87, cy + r * 0.5], [cx + r * 0.87, cy + r * 0.5]];
    const [a, b, c] = pts;
    const area = (p, q, o) => (q[0] - p[0]) * (o[1] - p[1]) - (q[1] - p[1]) * (o[0] - p[0]);
    for (let y = cy - r; y <= cy + r; y++)
      for (let x = cx - r * 0.9; x <= cx + r * 0.9; x++) {
        const d1 = area(a, b, [x, y]), d2 = area(b, c, [x, y]), d3 = area(c, a, [x, y]);
        const neg = d1 < 0 || d2 < 0 || d3 < 0, pos = d1 > 0 || d2 > 0 || d3 > 0;
        if (!(neg && pos)) set(x, y, 0);
      }
  };
  for (const a of anchors) tri(a.x * s, a.y * s, anchorR);
```

```js
import { lattice, SPACE } from './lattice.js';
import { bitsToPayload } from './codec.js';

function toGray(img) {
  const { width: w, height: h, data } = img;
  const g = new Uint8Array(w * h);
  for (let i = 0; i < w * h; i++) g[i] = (data[i * 4] * 0.3 + data[i * 4 + 1] * 0.59 + data[i * 4 + 2] * 0.11) | 0;
  return g;
}
function otsu(g) {
  const hist = new Array(256).fill(0);
  for (const v of g) hist[v]++;
  const total = g.length;
  let sum = 0; for (let i = 0; i < 256; i++) sum += i * hist[i];
  let sumB = 0, wB = 0, max = 0, thr = 127;
  for (let i = 0; i < 256; i++) {
    wB += hist[i]; if (!wB) continue;
    const wF = total - wB; if (!wF) break;
    sumB += i * hist[i];
    const mB = sumB / wB, mF = (sum - sumB) / wF;
    const between = wB * wF * (mB - mF) * (mB - mF);
    if (between > max) { max = between; thr = i; }
  }
  return thr;
}
// 8-connected components of dark pixels
function components(bin, w, h) {
  const label = new Int32Array(w * h).fill(-1);
  const blobs = [];
  const stack = [];
  for (let i = 0; i < w * h; i++) {
    if (bin[i] === 0 || label[i] !== -1) continue;
    const id = blobs.length;
    let area = 0, sx = 0, sy = 0, minx = w, miny = h, maxx = 0, maxy = 0;
    stack.push(i); label[i] = id;
    while (stack.length) {
      const p = stack.pop(); const px = p % w, py = (p / w) | 0;
      area++; sx += px; sy += py;
      if (px < minx) minx = px; if (px > maxx) maxx = px;
      if (py < miny) miny = py; if (py > maxy) maxy = py;
      for (let dy = -1; dy <= 1; dy++)
        for (let dx = -1; dx <= 1; dx++) {
          const nx = px + dx, ny = py + dy;
          if (nx < 0 || ny < 0 || nx >= w || ny >= h) continue;
          const q = ny * w + nx;
          if (bin[q] === 1 && label[q] === -1) { label[q] = id; stack.push(q); }
        }
    }
    const bw = maxx - minx + 1, bh = maxy - miny + 1;
    blobs.push({ area, cx: sx / area, cy: sy / area, fill: area / (bw * bh), bw, bh });
  }
  return blobs;
}
function solveAffine(src, dst) {
  // src/dst: 3 points each. Solve [a b c; d e f] mapping src->dst.
  const [s0, s1, s2] = src, [d0, d1, d2] = dst;
  const det = (s1.x - s0.x) * (s2.y - s0.y) - (s2.x - s0.x) * (s1.y - s0.y);
  if (Math.abs(det) < 1e-6) return null;
  const inv = 1 / det;
  const m11 = (s2.y - s0.y) * inv, m12 = -(s1.y - s0.y) * inv;
  const m21 = -(s2.x - s0.x) * inv, m22 = (s1.x - s0.x) * inv;
  const map = (p) => {
    const dx = p.x - s0.x, dy = p.y - s0.y;
    const u = m11 * dx + m12 * dy, v = m21 * dx + m22 * dy;
    return {
      x: d0.x + u * (d1.x - d0.x) + v * (d2.x - d0.x),
      y: d0.y + u * (d1.y - d0.y) + v * (d2.y - d0.y),
    };
  };
  return map;
}
function labelAnchors(cand, canon) {
  const d = (p, q) => Math.hypot(p.cx !== undefined ? p.cx : p.x, 0) && Math.hypot((p.cx ?? p.x) - (q.cx ?? q.x), (p.cy ?? p.y) - (q.cy ?? q.y));
  const perms = [[0,1,2],[0,2,1],[1,0,2],[1,2,0],[2,0,1],[2,1,0]];
  const cd = [d(canon[0], canon[1]), d(canon[1], canon[2]), d(canon[0], canon[2])];
  let best = null, bestErr = Infinity;
  for (const p of perms) {
    const a = cand[p[0]], b = cand[p[1]], c = cand[p[2]];
    const dd = [d(a, b), d(b, c), d(a, c)];
    // compare distance ratios (scale invariant)
    const r1 = dd[0] / dd[1], r2 = dd[2] / dd[1];
    const cr1 = cd[0] / cd[1], cr2 = cd[2] / cd[1];
    const err = Math.abs(r1 - cr1) + Math.abs(r2 - cr2);
    if (err < bestErr) { bestErr = err; best = [a, b, c]; }
  }
  return best.map((p) => ({ x: p.cx, y: p.cy }));
}

export function decode(img) {
  const { width: w, height: h } = img;
  const g = toGray(img);
  const thr = otsu(g);
  const bin = new Uint8Array(w * h);
  for (let i = 0; i < w * h; i++) bin[i] = g[i] < thr ? 1 : 0;

  const blobs = components(bin, w, h).filter((b) => b.area > 15);
  if (blobs.length < 3) return null;
  // anchors: 3 largest reasonably-solid blobs
  const anchorsCand = blobs.filter((b) => b.fill > 0.35).sort((a, b) => b.area - a.area).slice(0, 3);
  if (anchorsCand.length < 3) return null;

  const L = lattice();
  const canonAnchors = L.anchors.map((a) => ({ cx: a.x, cy: a.y }));
  const imgAnchors = labelAnchors(anchorsCand, canonAnchors);
  const map = solveAffine(L.anchors, imgAnchors);
  if (!map) return null;

  // sample each node
  const bits = new Uint8Array(256);
  const r = 4;
  L.nodes.forEach((n, idx) => {
    const p = map(n);
    let sum = 0, cnt = 0;
    for (let dy = -r; dy <= r; dy++)
      for (let dx = -r; dx <= r; dx++) {
        const x = Math.round(p.x + dx), y = Math.round(p.y + dy);
        if (x < 0 || y < 0 || x >= w || y >= h) continue;
        sum += g[y * w + x]; cnt++;
      }
    const mean = cnt ? sum / cnt : 255;
    bits[idx] = mean < thr ? 1 : 0; // dark center => solid => 1
  });
  return bitsToPayload(bits);
}
```

```js
export function blur(img, radius = 1) {
  const { width: w, height: h, data } = img;
  const out = new Uint8ClampedArray(data);
  const k = radius;
  for (let y = 0; y < h; y++)
    for (let x = 0; x < w; x++) {
      let s = 0, c = 0;
      for (let dy = -k; dy <= k; dy++)
        for (let dx = -k; dx <= k; dx++) {
          const nx = x + dx, ny = y + dy;
          if (nx < 0 || ny < 0 || nx >= w || ny >= h) continue;
          s += data[(ny * w + nx) * 4]; c++;
        }
      const v = s / c, o = (y * w + x) * 4;
      out[o] = out[o + 1] = out[o + 2] = v; out[o + 3] = 255;
    }
  return { width: w, height: h, data: out };
}
export function rotate(img, deg) {
  const { width: w, height: h, data } = img;
  const out = new Uint8ClampedArray(w * h * 4).fill(255);
  const rad = (deg * Math.PI) / 180, cos = Math.cos(rad), sin = Math.sin(rad);
  const cx = w / 2, cy = h / 2;
  for (let y = 0; y < h; y++)
    for (let x = 0; x < w; x++) {
      const sx = Math.round(cos * (x - cx) + sin * (y - cy) + cx);
      const sy = Math.round(-sin * (x - cx) + cos * (y - cy) + cy);
      const o = (y * w + x) * 4;
      if (sx < 0 || sy < 0 || sx >= w || sy >= h) { out[o] = out[o + 1] = out[o + 2] = 255; out[o + 3] = 255; continue; }
      const so = (sy * w + sx) * 4;
      out[o] = data[so]; out[o + 1] = data[so + 1]; out[o + 2] = data[so + 2]; out[o + 3] = 255;
    }
  return { width: w, height: h, data: out };
}
export function scaleDown(img, factor) {
  const { width: w, height: h, data } = img;
  const nw = Math.max(1, Math.round(w / factor)), nh = Math.max(1, Math.round(h / factor));
  const small = new Uint8ClampedArray(nw * nh * 4);
  for (let y = 0; y < nh; y++)
    for (let x = 0; x < nw; x++) {
      const so = ((Math.floor(y * factor)) * w + Math.floor(x * factor)) * 4, o = (y * nw + x) * 4;
      small[o] = data[so]; small[o + 1] = data[so + 1]; small[o + 2] = data[so + 2]; small[o + 3] = 255;
    }
  // upsample back to w,h (nearest) so decode() sees the resolution loss
  const out = new Uint8ClampedArray(w * h * 4).fill(255);
  for (let y = 0; y < h; y++)
    for (let x = 0; x < w; x++) {
      const sx = Math.min(nw - 1, Math.floor(x / factor)), sy = Math.min(nh - 1, Math.floor(y / factor));
      const so = (sy * nw + sx) * 4, o = (y * w + x) * 4;
      out[o] = small[so]; out[o + 1] = small[so + 1]; out[o + 2] = small[so + 2]; out[o + 3] = 255;
    }
  return { width: w, height: h, data: out };
}
export function noise(img, amount, seed = 1) {
  let a = seed >>> 0;
  const rnd = () => ((a = (a + 0x6d2b79f5) | 0), ((a ^ (a >>> 15)) >>> 0) / 4294967296);
  const { width: w, height: h, data } = img;
  const out = new Uint8ClampedArray(data);
  for (let i = 0; i < w * h; i++) {
    const n = (rnd() * 2 - 1) * amount, o = i * 4;
    const v = Math.max(0, Math.min(255, out[o] + n));
    out[o] = out[o + 1] = out[o + 2] = v;
  }
  return { width: w, height: h, data: out };
}
```
